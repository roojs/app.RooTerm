```vala
				if (page.get_data<Gtk.Widget>("tab-row").has_css_class("closing")) {

					return;
				}
				this.view.close_page(page);
			});
```

#### Keep

```vala
			var row = new Gtk.Box(Gtk.Orientation.HORIZONTAL, 0) {
				hexpand = false
			};
			row.add_css_class("host-tab");
			var hover = new Gtk.EventControllerMotion();
			hover.enter.connect(() => {
				row.add_css_class("hover");
			});
			hover.leave.connect(() => {
				row.remove_css_class("hover");
			});
			row.add_controller(hover);
			page.set_data("tab-row", row);

				row.add_css_class("selected");

			}
```

#### Remove

```vala

			row.append(close);
```

#### Replace with — one orange fill under Overlay; face = pick + close

```vala
			var fill = new Gtk.Box(Gtk.Orientation.HORIZONTAL, 0) {
				halign = Gtk.Align.START,
				valign = Gtk.Align.FILL,
				visible = false,
				width_request = 1
			};
			fill.add_css_class("host-tab-close-fill");
			var face = new Gtk.Box(Gtk.Orientation.HORIZONTAL, 0) {
				hexpand = false
			};
			var overlay = new Gtk.Overlay();
			overlay.child = fill;
			overlay.add_overlay(face);
			overlay.set_measure_overlay(face, true);
			face.append(pick);
			face.append(close);
			row.append(overlay);
```

#### Add — after `row.append(overlay);`: show fill + pause; width from countdown

```vala
			((Terminal.Base) page.child).closing.connect((left, total) => {
				if (left <= 0 || total <= 0) {
					row.remove_css_class("closing");
					fill.visible = false;
					fill.width_request = 1;
					close.icon_name = "window-close-symbolic";
					close.tooltip_text = "Close";
					return;
				}
				row.add_css_class("closing");
				fill.visible = true;
				fill.width_request = (this.tab_width * left) / total;
				close.icon_name = "media-playback-pause-symbolic";
				close.tooltip_text = "Cancel close";
			});
```
